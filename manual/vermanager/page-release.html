🚀 새 릴리즈 실행 (pages/release.php)

4단계 릴리즈 프로세스를 제공하는 핵심 UI 페이지입니다. 릴리즈 유형 선택, 소스 개선 연결, 릴리즈 노트 작성, 최종 실행까지 순차적으로 진행됩니다.

doRelease() JavaScript (완전 추적)

async function doRelease() {
    // 1) 선택된 릴리즈 타입 확인
    const type = document.querySelector('input[name="releaseType"]:checked').value;
    // 2) 릴리즈 노트 텍스트
    const note = document.getElementById('releaseNote').value.trim();
    // 3) 체크된 소스 개선 ID 수집
    const linked = Array.from(document.querySelectorAll('.imp-checkbox:checked'))
                      .map(cb => cb.value);

    // 4) 확인 대화상자
    const msg = `🚀 "${type.toUpperCase()}" 버전으로 릴리즈하시겠습니까?\n\n` +
        `대상: kiam.kr (운영) (kiam.kr)\n` +
        `연결된 소스 개선: ${linked.length}건\n` +
        `릴리즈 노트: ${note || '(없음)'}`;
    if (!confirm(msg)) return;

    // 5) 버튼 비활성화 + 로딩 상태
    const btn = document.querySelector('.vm-btn-lg');
    btn.disabled = true;
    btn.textContent = '⏳ 처리 중...';

    try {
        // 6) API 호출
        const res = await fetch('/vermanager/api/release', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ site: '${site_id}', type, note, linked_improvements: linked })
        });
        const data = await res.json();

        // 7) 결과 표시
        const resultDiv = document.getElementById('releaseResult');
        resultDiv.style.display = 'block';
        if (data.ok) {
            resultDiv.innerHTML = `✅ ${data.message}\n새 버전: v${data.new_version.raw}`;
            resultDiv.className = 'vm-result vm-result-success';
            setTimeout(() => { window.location.href = '?site=${site_id}'; }, 2000);
        } else {
            resultDiv.innerHTML = `❌ ${data.error}`;
            resultDiv.className = 'vm-result vm-result-error';
        }
    } catch (err) {
        // 네트워크 오류 처리
    } finally {
        btn.disabled = false;
        btn.textContent = '🚀 버전 릴리즈 실행';
    }
}

사이드바 네비게이션

릴리즈 페이지는 다른 페이지보다 확장된 사이드바를 제공합니다. 서버 관리와 Git 자동화 링크가 추가로 표시됩니다.

vm-nav:
  📊 대시보드
  🚀 새 릴리즈 (active)
  📜 버전 히스토리
  🔧 소스 개선 목록
  📝 CHANGELOG
  🔍 사이트 상세
  ────── (divider)
  🌐 사이트 관리
  🖥️ 서버 관리      ← 릴리즈 페이지에서 추가
  📦 Git 자동화     ← 릴리즈 페이지에서 추가
💡 핵심 포인트: 소스 개선 체크리스트는 improvements.json에서 released_in이 null인 항목만 표시합니다. 릴리즈 성공 시 2초 후 자동으로 대시보드로 리디렉션됩니다. SRC_IMPROVEMENTS_URL 상수를 통해 관리자 소스 개선 페이지로 바로 이동할 수 있습니다.
