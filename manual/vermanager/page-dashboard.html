데이터 로딩 (서버 로직 추적)

// 1) 시스템 통계 (site-registry.json 기반)
$stats = get_system_stats();   // → total_sites, production_sites, development_sites, total_servers

// 2) 현재 사이트 정보
$site_id = $current_site_id;   // → index.php에서 get_current_site_id() 결과
$site = $current_site;          // → get_site($site_id) 결과

// 3) 현재 사이트 버전 (package.json)
$version = site_get_current_version($site_id);

// 4) 버전 히스토리 (version_history.json, 역순 정렬)
$history = site_get_history($site_id);
$history = array_reverse($history);   // 최신순으로 뒤집기
$totalReleases = count($history);
$lastRelease = $totalReleases > 0 ? $history[0] : null;

// 5) 소스 개선 목록 (improvements.json)
$improvements = site_get_improvements($site_id);
$totalImprovements = count($improvements);
$unreleasedImps = count(array_filter($improvements, fn($i) => empty($i['released_in'])));

// 6) CHANGELOG
$changelog = site_get_changelog($site_id);

// 7) 사이트별 요약 (모든 활성 사이트의 버전+릴리즈수)
$site_summaries = [];
foreach ($all_sites as $sid => $s) {
    $sv = site_get_current_version($sid);
    $sh = site_get_history($sid);
    $site_summaries[] = ['id'=>$sid, 'name'=>..., 'version'=>$sv['raw'], 'releases'=>count($sh)...];
}

// 8) 다음 버전 미리보기
$nextPatch = bump_version($version, 'patch');    // v1.0.1 → v1.0.2
$nextMinor = bump_version($version, 'minor');    // v1.0.1 → v1.1.0
$nextMajor = bump_version($version, 'major');    // v1.0.1 → v2.0.0

빠른 릴리즈 JavaScript

// 사이트 전환
function switchSite(siteId) {
    window.location.href = '?site=' + siteId;
}

// 빠른 릴리즈
async function quickRelease(type) {
    const res = await fetch('/vermanager/api/release', {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({
            site: '',
            type: type,
            note: '대시보드 빠른 릴리즈'
        })
    });
    const data = await res.json();
    if (data.ok) {
        alert('✅ ' + data.message);
        location.reload();   // 대시보드 갱신
    }
}
💡 핵심 포인트: 대시보드는 페이지 로딩 시 4개의 JSON 파일(package.json, version_history.json, improvements.json, changelog.json)을 모두 읽습니다. 사이트 선택기 드롭다운으로 운영↔개발 환경을 즉시 전환할 수 있습니다. 빠른 릴리즈는 확인 대화상자 없이 즉시 PATCH 버전을 증가시킵니다.
