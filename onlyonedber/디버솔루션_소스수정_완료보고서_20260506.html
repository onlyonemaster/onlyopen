6.3 GitHub Actions CI/CD (권장)

name: Build OnlyoneDBer
on: [push, workflow_dispatch]
jobs:
  build:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-dotnet@v4
        with: { dotnet-version: '8.0.x' }
      - name: NuGet 복원
        run: nuget restore OnlyoneDBer.sln
      - name: Release 빌드
        run: msbuild OnlyoneDBer.sln /p:Configuration=Release /p:Platform=x86
      - name: 빌드 결과물 업로드
        uses: actions/upload-artifact@v4
        with:
          name: onlyonedber-release
          path: |
            **/bin/Release/**/*.exe
            **/bin/Release/**/*.dll
            **/bin/Release/**/*.config
